import nltk
nltk.download('vader_lexicon')
from nltk.sentiment.vader import SentimentIntensityAnalyzer

sia = SentimentIntensityAnalyzer()

narratives_for_sentiment = complaints_df["Consumer complaint narrative"].dropna().reset_index(drop=True)

sentiment_scores = narratives_for_sentiment.apply(lambda x: sia.polarity_scores(x)['compound'])

complaints_sentiment = complaints_df.dropna(subset=["Consumer complaint narrative"]).copy()
complaints_sentiment["sentiment_score"] = sentiment_scores.values

def categorize_sentiment(score):
    if score >= 0.05:
        return "Positive"
    elif score <= -0.05:
        return "Negative"
    else:
        return "Neutral"

complaints_sentiment["sentiment_category"] = complaints_sentiment["sentiment_score"].apply(categorize_sentiment)
